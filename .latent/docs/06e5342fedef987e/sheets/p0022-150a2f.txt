// Initialize Time
configTime(19800, 0, "pool.ntp.org"); // IST: GMT+5:30
while(!time(nullptr)) delay(1000);
// Initialize BLE
BLEDevice::init("");
pBLEScan = BLEDevice::getScan();
pBLEScan->setActiveScan(true);
pBLEScan->setInterval(97);
pBLEScan->setWindow(37);
}
void loop() {
checkDailyReset();
processBLEScan();
checkForExits();
delay(2000);
}
void processBLEScan() {
BLEScanResults* scanResults = pBLEScan->start(SCAN_TIME, false); // Fixed pointer declaration
time_t now;
time(&now);
if (scanResults) {
for(int i = 0; i < scanResults->getCount(); i++) { // Use -> operator
BLEAdvertisedDevice device = scanResults->getDevice(i);
String devName = device.getName().c_str();
for(auto &emp : employees) {
if(devName == emp.beaconName) {
if (!emp.isActive) { // Entry event
emp.isActive = true;
emp.entryTime = now;
logFirebaseEvent(emp.empID, "entry", now);
}
emp.lastSeen = now; // Update last seen time
}
}
}
pBLEScan->clearResults();
}
}
void checkForExits() {
time_t now;
time(&now);
for (auto &emp : employees) {
if (emp.isActive && (now - emp.lastSeen > EXIT_TIMEOUT)) {
emp.isActive = false;
emp.dailyTotal += (now - emp.entryTime);
logFirebaseEvent(emp.empID, "exit", now);
Firebase.setDouble(fbdo, getFirebasePath(emp.empID) + "/total", emp.dailyTotal);
}
}
}
